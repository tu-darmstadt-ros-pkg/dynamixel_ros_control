# LeakSanitizer suppressions for the dynamixel_ros_control test suite.
#
# Allocations that survive into process exit are reported by LSan as leaks.
# The test main calls _Exit() to bypass the racy ROS 2 / Cyclone DDS teardown
# (see test_hardware_interface_common.hpp), which means the rclcpp / rcl /
# DDS infrastructure is still holding live allocations when LSan runs. Those
# allocations are owned and tracked by ROS 2 itself, not by our code.
#
# This file suppresses the well-known framework-side allocations so real
# leaks in dynamixel_ros_control still surface. Each entry matches by
# function name in any frame of the leak's allocation stack.
#
# When adding a suppression, prefer the highest-level frame that is
# unambiguously framework-owned so we don't accidentally hide leaks in our
# own code that happen to call through these APIs.

# rcl: nodes, publishers, services, type cache, type-description service.
leak:rcl_publisher_init
leak:rcl_subscription_init
leak:rcl_service_init
leak:rcl_client_init
leak:rcl_node_init
leak:rcl_logging_rosout_init_publisher_for_node
leak:rcl_node_type_cache_register_type
leak:rcl_node_type_description_service_init
leak:rcl_convert_type_source_sequence_runtime_to_msg

# rclcpp: node and node-interface construction.
leak:rclcpp::Node::Node
leak:rclcpp::node_interfaces::NodeBase::NodeBase
leak:rclcpp::node_interfaces::NodeParameters::NodeParameters
leak:rclcpp::node_interfaces::NodeTypeDescriptions::NodeTypeDescriptions
leak:rclcpp::ParameterService::ParameterService
leak:rclcpp::executors::

# Cyclone DDS / rmw plugin: receiver threads and DDS reader/writer entities
# kept alive past _Exit().
leak:libddsc.so
leak:librmw_cyclonedds_cpp.so
leak:librmw_implementation.so
leak:rmw_create_node
leak:rmw_create_publisher
leak:rmw_create_subscription
leak:rmw_create_service
leak:rmw_create_client

# ros2_control: hardware component initialization creates a Node per component.
# The leaks are owned by hardware_interface itself, not by our DynamixelHardwareInterface.
leak:hardware_interface::HardwareComponentInterface::init
leak:hardware_interface::HardwareComponent::initialize
leak:hardware_interface::ResourceStorage::

# controller_manager: spawns its own helper nodes.
leak:controller_manager::ControllerManager

# hector_testing_utils: TestNode and TestClient instances owned by the gtest fixture.
# They are destroyed in HectorTestFixture::TearDown, but _Exit() may skip the
# final fixture destructor — these allocations are reachable, not lost.
leak:hector_testing_utils::TestNode
leak:hector_testing_utils::TestClient
leak:hector_testing_utils::TestExecutor
leak:hector_testing_utils::HectorTestFixture::SetUp
