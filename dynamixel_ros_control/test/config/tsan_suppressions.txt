# ThreadSanitizer suppressions for the dynamixel_ros_control test suite.
#
# Each entry suppresses a class of reports keyed by a substring of a frame
# in the report's stack. We aim to suppress only known-framework-side races
# so real races in dynamixel_ros_control surface clearly.
#
# Format:
#   race:<symbol_or_substring>
#   thread:<symbol>
#   mutex:<symbol>
#   deadlock:<symbol>

# rclcpp executor / wait-set internals — known races between executor threads
# that don't propagate to user code.
race:rclcpp::executors::ExecutorNotifyWaitable
race:rclcpp::executors::Executor
race:rcl_wait_set_

# Cyclone DDS receiver / writer internals. Reports surface as raw libddsc.so
# offsets because the library doesn't ship debug symbols on the stock install.
called_from_lib:libddsc.so
called_from_lib:librmw_cyclonedds_cpp.so

# Test fixture's update loop uses simulated time / std::chrono interactions
# that rclcpp::Clock races against on shutdown — owned by rclcpp, not us.
race:rclcpp::Clock
race:rclcpp::TimeSource

# Spurious tsan reports inside the sanitizer's own malloc/free interceptors
# triggered by ROS shutdown ordering (operator delete[] in tsan_new_delete.cpp
# called from rclcpp / DDS). These never identify a real userland bug.
race:operator delete[]
